```jsx
function CourseCard({ course }) {
  return (
    <article>
      <h2>{course.title}</h2>
      <p>Duration: {course.duration}</p>

      <p>
        {course.isOpen
          ? "Enrolment Open"
          : "Enrolment Closed"}
      </p>
    </article>
  );
}

function CourseList({ courses }) {
  if (courses.length === 0) {
    return <p>No courses are available.</p>;
  }

  return (
    <div>
      {courses.map((course) => (
        <CourseCard
          key={course.id}
          course={course}
        />
      ))}
    </div>
  );
}

export default CourseList;
```
